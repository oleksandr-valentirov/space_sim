# Прапорці компіляції ядра. ЄДИНЕ місце, де вони задані.
#
# Читає і Makefile (M0), і core-sys/build.rs (з M1, крок D1 у ROADMAP.md).
# Розбіжність між двома збірками тихо ламає детермінізм — PROJECT.md §4.
# Рядки з # ігноруються.

-std=c11

# Попередження = помилки. PROJECT.md §9.
-Wall
-Wextra
-Werror

-O2
-g

# Детермінізм. Забороняємо компілятору зливати множення й додавання в FMA:
# інакше той самий вихідний код дає різні біти на різних платформах.
# -ffast-math НІКОЛИ, за жодних обставин. PROJECT.md §4.
-ffp-contract=off

# Дозволяє компілятору не викликати libm заради errno і брати апаратну
# інструкцію sqrt напряму. Без цього sqrt не злінкується (ми свідомо не
# лінкуємо -lm), а з -lm у ядро просочився б увесь libm.
#
# На бітову точність НЕ впливає: IEEE-754 вимагає коректного заокруглення
# sqrt, тож значення те саме, зникає лише запис у errno. Це окремий прапорець,
# а не частина -ffast-math у сенсі переупорядкування — реасоціації він
# не дозволяє.
-fno-math-errno
